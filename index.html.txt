                '</div>';
            slide.querySelector('.btn-watch').addEventListener('click', function(e) {

                openModal(featuredItems[idx]);
            });
            track.appendChild(slide);
            var dot = document.createElement('div');
            dot.className = 'slider-dot' + (i === 0 ? ' active' : '');
            dot.addEventListener('click', (function(n){ return function(){ goToSlide(n); }; })(i));
            dotsWrap.appendChild(dot);

        document.querySelector('.hero').addEventListener('mouseenter', function(){ clearInterval(autoSlideTimer); });
        document.querySelector('.hero').addEventListener('mouseleave', startAutoSlide);
    }

    function goToSlide(index) {
        var slides = document.querySelectorAll('.slide');
        var dots = document.querySelectorAll('.slider-dot');
        slides[currentSlide].classList.remove('active');
        dots[currentSlide].classList.remove('active');
        currentSlide = ((index % slides.length) + slides.length) % slides.length;
        slides[currentSlide].classList.add('active');
        dots[currentSlide].classList.add('active');
        document.getElementById('sliderTrack').style.transform = 'translateX(-' + (currentSlide * 100) + '%)';
    }

    window.changeSlide = function(dir) { goToSlide(currentSlide + dir); };

    function startAutoSlide() {
        clearInterval(autoSlideTimer);
        autoSlideTimer = setInterval(function(){ goToSlide(currentSlide + 1); }, 6000);
    }

    function playVideo(title, url, year, category, type, epNum, seasonName) {
        _shareEpNum = epNum;
        _shareSeasonName = seasonName;

        var frame = document.getElementById('videoFrame');
        var loader = document.getElementById('videoLoader');
        var titleEl = document.getElementById('modalTitle');

        if (seasonName && epNum) { displayTitle = title + ' — ' + seasonName + ' — الحلقة ' + epNum; }
        else if (epNum) { displayTitle = title + ' — الحلقة ' + epNum; }
        titleEl.textContent = displayTitle;

        frame.src = '';

        document.body.style.overflow = 'hidden';

    function buildSeasonsBar(series) {
        var bar = document.getElementById('seasonsBar');
        bar.innerHTML = '';
        if (series.seasons.length <= 1) { bar.style.display = 'none'; return; }
        bar.style.display = 'flex';
        var lbl = document.createElement('span');
        lbl.className = 'seasons-label';
        lbl.textContent = 'المواسم:';
        bar.appendChild(lbl);
        series.seasons.forEach(function(season, idx) {
            var tab = document.createElement('button');
            tab.className = 'season-tab' + (idx === 0 ? ' active' : '');
            tab.innerHTML = season.name + ' <span class="season-ep-count">' + season.eps.length + '</span>';
            tab.addEventListener('click', (function(i, t) {
                return function() {
                    document.querySelectorAll('.season-tab').forEach(function(b){ b.classList.remove('active'); });
                    t.classList.add('active');
                    _activeSeason = i;
                    renderEpisodes(series, i);
                };
            })(idx, tab));
            bar.appendChild(tab);
        });
    }

    function renderEpisodes(series, seasonIdx) {
        var season = series.seasons[seasonIdx];
        var grid = document.getElementById('episodesGrid');
        var key = series.title + '_s' + seasonIdx;
        if (!_watched[key]) _watched[key] = {};
        grid.style.opacity = '0';
        setTimeout(function() {
            grid.innerHTML = '';
            season.eps.forEach(function(epUrl, idx) {
                var epNum = idx + 1;
                var btn = document.createElement('button');
                btn.className = 'ep-btn' + (_watched[key][epNum] ? ' watched' : '');
                btn.innerHTML = '<span>' + epNum + '</span><span class="ep-label">حلقة</span>';
                btn.addEventListener('click', (function(n, u, b) {
                    return function() {
                        _watched[key][n] = true;
                        b.classList.add('watched');
                        _currentShareMovie = series;
                        _shareEpNum = n;
                        _shareSeasonName = series.seasons.length > 1 ? season.name : null;
                        closeEpisodesModal();
                        var sn = series.seasons.length > 1 ? season.name : null;
                        setTimeout(function() {
                            playVideo(series.title, u, series.year, series.category, series.type, n, sn);
                        }, 350);
                    };
                })(epNum, epUrl, btn));
                grid.appendChild(btn);
            });
            grid.style.transition = 'opacity 0.2s';
            grid.style.opacity = '1';
        }, 120);
    }

    window.closeEpisodesModal = function() {
        document.getElementById('episodesModal').classList.remove('open');
        document.body.style.overflow = '';
    };

    function buildShareData(movie, epNum, seasonName) {
        var metaParts = [movie.category];
        var epLine = '';
        if (movie.type === 'مسلسلات') {
            if (seasonName) { metaParts.push(seasonName); }
            epLine = epNum ? ('الحلقة ' + epNum) : 'مسلسل';
        } else {
            epLine = 'فيلم';
        }
        var metaLine = metaParts.join(' · ');
        var shareText = 'شاهد الآن على VÉRANO VIP\n\n';
        shareText += '🎬 ' + movie.title + '\n';
        shareText += '📅 ' + movie.year + ' · ' + metaLine + '\n';
        shareText += (movie.type === 'مسلسلات' && epNum) ? ('📺 ' + epLine + '\n') : ('🎞 ' + epLine + '\n');
        shareText += '\n' + movie.description + '\n\n';
        shareText += '▶ VÉRANO VIP';
        return {
            title: movie.title,
            poster: movie.poster,
            year: movie.year,
            category: movie.category,
            type: movie.type,
            metaLine: metaLine,
            epLine: epLine,
            description: movie.description,
            shareText: shareText
        };
    }

    function openSharePopup(data) {
        _shareData = data;
        var loading = document.getElementById('shareCanvasLoading');
        loading.classList.remove('hidden');
        document.getElementById('sharePopup').classList.add('open');
        var copyBtn = document.getElementById('shareCopyBtn');
        copyBtn.textContent = '📋 نسخ النص';
        copyBtn.classList.remove('copied');

        var img = new Image();
        img.crossOrigin = 'anonymous';

        img.onload = function() {
            drawShareCard(img, data);
            loading.classList.add('hidden');
        };
        img.onerror = function() {
            var proxy = 'https:' + '//' + 'api.allorigins.win/raw?url=' + encodeURIComponent(data.poster);
            var img2 = new Image();
            img2.crossOrigin = 'anonymous';
            img2.onload = function() { drawShareCard(img2, data); loading.classList.add('hidden'); };
            img2.onerror = function() { drawShareCard(null, data); loading.classList.add('hidden'); };
            img2.src = proxy;
        };
        img.src = data.poster + (data.poster.indexOf('?') === -1 ? '?' : '&') + '_t=' + Date.now();
    }

    function drawShareCard(posterImg, data) {
        var W = 1080, H = 1080;
        var canvas = document.getElementById('shareCanvas');
        canvas.width = W; canvas.height = H;
        var ctx = canvas.getContext('2d');

        ctx.fillStyle = '#050507';
        ctx.fillRect(0, 0, W, H);

        if (posterImg) {
            var aspect = posterImg.naturalWidth / posterImg.naturalHeight;
            var ph = H, pw = ph * aspect;
            var px = (W - pw) / 2;
            ctx.drawImage(posterImg, px, 0, pw, ph);
            var grad = ctx.createLinearGradient(0, H * 0.3, 0, H);
            grad.addColorStop(0, 'rgba(5,5,7,0)');
            grad.addColorStop(0.5, 'rgba(5,5,7,0.85)');
            grad.addColorStop(1, 'rgba(5,5,7,1)');
            ctx.fillStyle = grad;
            ctx.fillRect(0, 0, W, H);
            var grad2 = ctx.createLinearGradient(0, 0, W, 0);
            grad2.addColorStop(0, 'rgba(5,5,7,0.5)');
            grad2.addColorStop(1, 'rgba(5,5,7,0)');
            ctx.fillStyle = grad2;
            ctx.fillRect(0, 0, W, H);
        }

        var PAD = 60;
        var maxW = W - PAD * 2;

        function wrapText(text, font, maxWidth, maxLines) {
            ctx.font = font;
            var words = (text || '').split(' ');
            var lines = [], cur = '';
            for (var i = 0; i < words.length; i++) {
                var trial = cur ? cur + ' ' + words[i] : words[i];
                if (ctx.measureText(trial).width > maxWidth && cur) { lines.push(cur); cur = words[i]; }
                else { cur = trial; }
            }
            if (cur) lines.push(cur);
            return lines.slice(0, maxLines);
        }

        ctx.textAlign = 'right';
        var RX = W - PAD;

        ctx.save();
        ctx.shadowColor = 'rgba(201,168,76,0.6)';
        ctx.shadowBlur = 30;
        ctx.fillStyle = '#E8C96A';
        ctx.font = 'bold 52px serif';
        ctx.fillText('VÉRANO VIP', RX, 80);
        ctx.restore();

        ctx.fillStyle = 'rgba(201,168,76,0.5)';

        ctx.fillStyle = '#C9A84C';
        var badge = (data.category || '') + (data.type ? '  \u00b7  ' + data.type : '');
        ctx.fillText(badge, RX, 158);

        if (data.epLine) {
            ctx.font = 'bold 28px sans-serif';
            ctx.fillStyle = 'rgba(201,168,76,0.8)';
            ctx.fillText(data.epLine, RX, 205);
        }

        ctx.shadowColor = 'rgba(0,0,0,0.9)';
        ctx.shadowBlur = 20;
        ctx.fillStyle = '#ffffff';
        var titleLines = wrapText(data.title || '', 'bold 72px sans-serif', maxW, 2);
        var titleY = H - 340;
        for (var li = 0; li < titleLines.length; li++) {
            ctx.font = 'bold 72px sans-serif';
            ctx.fillText(titleLines[li], RX, titleY + li * 88);
        }
        ctx.shadowBlur = 0;

        var desc = data.description || '';
        if (desc.length > 90) desc = desc.substring(0, 90) + '...';
        var descLines = wrapText(desc, '32px sans-serif', maxW, 2);
        ctx.fillStyle = 'rgba(240,235,224,0.78)';
        var descY = H - 175;
        for (var dli = 0; dli < descLines.length; dli++) {
            ctx.font = '32px sans-serif';
            ctx.fillText(descLines[dli], RX, descY + dli * 46);
        }

        ctx.fillStyle = 'rgba(201,168,76,0.4)';
        ctx.fillRect(PAD, H - 100, maxW, 1);

        ctx.font = 'bold 30px sans-serif';
        ctx.fillStyle = '#C9A84C';
        ctx.fillText((data.year || '') + (data.metaLine ? '  \u00b7  ' + data.metaLine : ''), RX, H - 55);
    }
